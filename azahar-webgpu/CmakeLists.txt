cmake_minimum_required(VERSION 3.18)
project(AzaharWebGPU VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Always enable WebGPU and Azahar
set(USE_WEBGPU ON)
set(INTEGRATE_AZAHAR ON)

list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")

# -------------------------------
# WebGPU Core
# -------------------------------
add_library(webgpu_core STATIC
    src/webgpu_core/webgpu_device.cpp
    src/webgpu_core/webgpu_device.h
    src/webgpu_core/webgpu_buffer.cpp
    src/webgpu_core/webgpu_buffer.h
    src/webgpu_core/webgpu_shader.cpp
    src/webgpu_core/webgpu_shader.h
    src/webgpu_core/webgpu_texture.cpp
    src/webgpu_core/webgpu_texture.h
    src/webgpu_core/webgpu_pipeline.cpp
    src/webgpu_core/webgpu_pipeline.h
)

target_include_directories(webgpu_core PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}/src
    # fmt is supplied by the upstream Azahar checkout rather than copied into
    # this overlay. Keep the overlay's own include directory first so its
    # WebGPU-specific logging types remain authoritative.
    ${CMAKE_CURRENT_SOURCE_DIR}/../azahar/src
    ${CMAKE_CURRENT_SOURCE_DIR}/../azahar/externals/fmt/include
)

if(EMSCRIPTEN)
    target_compile_options(webgpu_core PUBLIC -sUSE_WEBGPU=1)
    target_link_options(webgpu_core PUBLIC -sUSE_WEBGPU=1)
else()
    target_link_libraries(webgpu_core PUBLIC ${WEBGPU_LIBRARIES})
    target_include_directories(webgpu_core PUBLIC ${WEBGPU_INCLUDE_DIRS})
endif()

# -------------------------------
# WebGPU Renderer
# -------------------------------
add_library(webgpu_renderer STATIC
    src/renderer/renderer_webgpu.cpp
    src/renderer/renderer_webgpu.h
)

target_link_libraries(webgpu_renderer PUBLIC webgpu_core)

# -------------------------------
# Azahar Emulator Core
# -------------------------------
# Always include, even for web builds

add_subdirectory(../azahar azahar)

# Ensure Azahar links to our renderer
if(TARGET citra_core)
    target_link_libraries(citra_core PUBLIC webgpu_renderer)
endif()

# -------------------------------
# Examples / Main
# -------------------------------
add_executable(azahar_webgpu examples/main.cpp)  # your main that boots Azahar in browser
target_link_libraries(azahar_webgpu PUBLIC citra_core)
if(EMSCRIPTEN)
    target_compile_options(azahar_webgpu PUBLIC -sUSE_WEBGPU=1)
    target_link_options(azahar_webgpu PRIVATE
        -sINITIAL_MEMORY=512MB
        -sSTACK_SIZE=2MB
        -sALLOW_MEMORY_GROWTH=1
        -sNO_EXIT_RUNTIME=1
        -sUSE_WEBGPU=1
    )
endif()
